# lumen_core: platform-independent foundation (fs, json, log, process, pak, lua compiler, project model).
file(GLOB LUMEN_CORE_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp)
add_library(lumen_core STATIC ${LUMEN_CORE_SOURCES})
add_library(lumen::core ALIAS lumen_core)
target_include_directories(lumen_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(lumen_core PUBLIC cxx_std_17)
set_target_properties(lumen_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)

if(TARGET LUA)
  target_link_libraries(lumen_core PUBLIC LUA)
  target_include_directories(lumen_core PRIVATE ${LUMEN_LUA_INCLUDE_DIR})
elseif(TARGET lumen_lua)
  target_link_libraries(lumen_core PUBLIC lumen_lua)
else()
  message(FATAL_ERROR "lumen_core needs a Lua target (Wicked's LUA, or lumen_lua for standalone builds)")
endif()

if(MSVC)
  target_compile_options(lumen_core PRIVATE /W4 /utf-8 /permissive- /GR- /EHs-c-)
  target_compile_definitions(lumen_core PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(lumen_core PRIVATE -Wall -Wextra -Wno-unused-parameter -fno-exceptions -fno-rtti)
endif()
if(UNIX AND NOT APPLE)
  target_link_libraries(lumen_core PUBLIC ${CMAKE_DL_LIBS} Threads::Threads)
endif()
if(WIN32)
  target_link_libraries(lumen_core PUBLIC shell32 ole32 advapi32)
endif()
